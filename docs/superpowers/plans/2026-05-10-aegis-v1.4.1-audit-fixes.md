### Plan note

`runStaticPass` runs analyzers via `coroutineScope { ... async ... awaitAll() }` — analyzers fan out in parallel. The PCE thrown by `pceAnalyzer` cancels the surrounding `coroutineScope` and propagates out of `awaitAll()`. The "subsequent analyzer not invoked" assertion is therefore racy — it depends on whether the recorder's coroutine had time to start. Drop that assertion; the meaningful guarantee is that PCE escapes the engine, which is what the audit finding C1 is about.

### Plan note

`Map<String, Any>` serialization with `kotlinx.serialization` requires `Json { ignoreUnknownKeys = true; encodeDefaults = true }` (already present in `JcefBridge`'s `json` field). The `Any`-valued map serializer works for primitives + strings + nested lists; for the `List<DebugVariable>` value, `DebugVariable` is already `@Serializable` — no additional annotation needed.

If `json.encodeToString(map)` doesn't compile because of `Any` ambiguity, switch to `JsonObject` / `JsonPrimitive` builders:

```kotlin
val payload = buildJsonObject {
    put("nodeId", JsonPrimitive(nodeId))
    put("status", JsonPrimitive(status.name))
}.toString()
```

Decide at compile time which approach the codebase accepts.

/**
 * Pure-JUnit canary that the JSON encoding used by JcefBridge survives
 * pathological inputs that the previous hand-built escapes broke on.
 *
 * Doesn't instantiate JcefBridge (which needs JCEF at runtime); instead
 * pins the exact `Json` config and serialization shape so a future change
 * that drops `encodeDefaults = true` etc. fails this test, not the user.
 */
class JcefBridgeJsonEncodingTest {

    private val json = Json { ignoreUnknownKeys = true; encodeDefaults = true }

    @Test fun `backslash in nodeId is escaped`() {
        val payload = json.encodeToString(mapOf("nodeId" to "C:\\Users\\foo", "status" to "ERROR"))
        assertEquals("""{"nodeId":"C:\\Users\\foo","status":"ERROR"}""", payload)
    }

    @Test fun `newline in error message is escaped`() {
        val payload = json.encodeToString(mapOf("message" to "line one\nline two"))
        assertTrue(payload.contains("\\n"), "Expected newline escape in: $payload")
        assertTrue(!payload.contains("\n"), "Expected no raw newline in: $payload")
    }

    @Test fun `tab and carriage return are escaped`() {
        val payload = json.encodeToString(mapOf("text" to "col1\tcol2\rdone"))
        assertTrue(payload.contains("\\t"), payload)
        assertTrue(payload.contains("\\r"), payload)
    }

    @Test fun `quote in payload is escaped`() {
        val payload = json.encodeToString(mapOf("title" to "she said \"hi\""))
        assertEquals("""{"title":"she said \"hi\""}""", payload)
    }

    @Test fun `unicode survives encoding`() {
        val payload = json.encodeToString(mapOf("name" to "café 日本語 ✓"))
        assertTrue(payload.contains("café"), payload)
        assertTrue(payload.contains("日本語"), payload)
    }

    @Test fun `null byte is escaped`() {
        val payload = json.encodeToString(mapOf("k" to "a b"))
        // kotlinx.serialization escapes   as  
        assertTrue(payload.contains("\\u0000"), payload)
    }

    @Test fun `payload is a valid JSON object`() {
        val raw = "}{<script>alert(1)</script>"
        val payload = json.encodeToString(mapOf("title" to raw))
        // The payload itself is valid JSON; round-tripping through Json.parseToJsonElement
        // must succeed and recover the original string.
        val parsed = Json.parseToJsonElement(payload)
        assertEquals(
            raw,
            (parsed as kotlinx.serialization.json.JsonObject)["title"]
                ?.let { (it as kotlinx.serialization.json.JsonPrimitive).content }
        )
    }
}

```kotlin
fun findCycles(): List<List<String>> {
    cachedCycles?.let { return it }
    val cycles = mutableListOf<List<String>>()
    val visited = mutableSetOf<String>()

    for (root in nodes.keys) {
        if (root in visited) continue

        // Tarjan-style iterative DFS using an explicit recursion stack.
        // Each frame carries the node and an iterator over its outgoing neighbors;
        // when the iterator exhausts, we pop the frame (post-order step).
        data class Frame(val node: String, val iter: Iterator<String>)

        val recursionStack = mutableSetOf<String>()
        val path = mutableListOf<String>()
        val stack = ArrayDeque<Frame>()

        visited.add(root)
        recursionStack.add(root)
        path.add(root)
        stack.addLast(Frame(root, (adjacencyList[root] ?: emptySet()).iterator()))

        while (stack.isNotEmpty()) {
            val top = stack.last()
            if (top.iter.hasNext()) {
                val neighbor = top.iter.next()
                if (neighbor !in visited) {
                    visited.add(neighbor)
                    recursionStack.add(neighbor)
                    path.add(neighbor)
                    stack.addLast(Frame(neighbor, (adjacencyList[neighbor] ?: emptySet()).iterator()))
                } else if (neighbor in recursionStack) {
                    val start = path.indexOf(neighbor)
                    if (start >= 0) cycles.add(path.subList(start, path.size).toList())
                }
            } else {
                recursionStack.remove(top.node)
                path.removeLast()
                stack.removeLast()
            }
        }
    }

    cachedCycles = cycles
    return cycles
}
```

/**
 * Stress test for InMemoryGraph adjacency-list thread safety. Pre-V1.4.1, the
 * outer ConcurrentHashMap nested a non-concurrent LinkedHashSet — concurrent
 * addEdge calls with a shared source could ConcurrentModificationException
 * or corrupt the inner set.

class InMemoryGraphConcurrencyTest {

    @Test fun `concurrent addEdge calls do not throw or lose targets`() {
        val graph = InMemoryGraph()
        val source = "src"
        graph.addNode(GraphNode(
            id = source, name = source, filePath = "/src.kt",
            type = NodeType.FILE, status = NodeStatus.HEALTHY, complexity = 0
        ))

        val threadCount = 8
        val edgesPerThread = 1000
        val totalEdges = threadCount * edgesPerThread
        for (i in 0 until totalEdges) {
            graph.addNode(GraphNode(
                id = "tgt-$i", name = "tgt-$i", filePath = "/t$i.kt",
                type = NodeType.FILE, status = NodeStatus.HEALTHY, complexity = 0
            ))
        }

        val pool = Executors.newFixedThreadPool(threadCount)
        val startGate = CountDownLatch(1)
        val doneGate = CountDownLatch(threadCount)
        val firstError = AtomicReference<Throwable?>(null)

        for (t in 0 until threadCount) {
            pool.submit {
                try {
                    startGate.await()
                    for (i in 0 until edgesPerThread) {
                        val targetIndex = t * edgesPerThread + i
                        graph.addEdge(GraphEdge(
                            id = "e-$targetIndex",
                            source = source,
                            target = "tgt-$targetIndex"
                        ))
                    }
                } catch (e: Throwable) {
                    firstError.compareAndSet(null, e)
                } finally {
                    doneGate.countDown()
                }
            }
        }

        startGate.countDown()  // release all threads simultaneously
        assertTrue(doneGate.await(30, TimeUnit.SECONDS), "Threads did not complete within 30s")
        pool.shutdown()

        firstError.get()?.let { throw AssertionError("Concurrent addEdge threw: $it", it) }

        // Every target should be reachable from source.
        val neighbors = graph.getNeighbors(source)
        assertEquals(totalEdges, neighbors.size,
            "Expected $totalEdges neighbors after concurrent inserts, got ${neighbors.size}")
    }

Add the override:

```kotlin
override fun generateFixFromPsi(issue: Issue, file: PsiFile): CodeFix? {
    val ktFile = file as? KtFile ?: return null
    val varName = titleVarRegex.find(issue.title)?.groupValues?.get(1) ?: return null
    val document = PsiDocumentManager.getInstance(file.project).getDocument(ktFile) ?: return null

    // Find the first KtDotQualifiedExpression on the issue line whose receiver is
    // varName and which isn't already a safe call.
    val target = PsiTreeUtil.findChildrenOfType(ktFile, KtDotQualifiedExpression::class.java)
        .firstOrNull { dot ->
            document.getLineNumber(dot.textOffset) + 1 == issue.line &&
                (dot.receiverExpression as? KtNameReferenceExpression)?.getReferencedName() == varName &&
                dot.operationTokenNode.elementType != KtTokens.SAFE_ACCESS
        } ?: return null

    val lineStart = document.getLineStartOffset(issue.line - 1)
    val lineEnd = document.getLineEndOffset(issue.line - 1)
    val original = document.getText(TextRange(lineStart, lineEnd))

    // Replace the dot operator (single character) at the operationTokenNode offset
    // with a safe-call. Compute the offset relative to the line start.
    val dotOffsetInLine = target.operationTokenNode.startOffset - lineStart
    val fixed = original.substring(0, dotOffsetInLine) + "?." +
                original.substring(dotOffsetInLine + 1)

    if (fixed == original) return null

    return CodeFix(
        id = java.util.UUID.randomUUID().toString(),
        issueId = issue.id,
        description = "Add optional chaining: $varName. → $varName?.",
        originalCode = original,
        fixedCode = fixed,
        filePath = issue.filePath,
        lineStart = issue.line,
        lineEnd = issue.line,
        isDeterministic = true,
        confidence = 1.0
    )
}
```

```kotlin
private fun stripInlineComments(line: String): String {
    // Strip // line-end comments
    val noLineComment = run {
        var inString = false
        var stringChar = ' '
        for (i in 0 until line.length - 1) {
            val c = line[i]
            if (!inString && (c == '"' || c == '\'' || c == '`')) {
                inString = true; stringChar = c
            } else if (inString && c == stringChar && (i == 0 || line[i - 1] != '\\')) {
                inString = false
            } else if (!inString && c == '/' && line[i + 1] == '/') {
                return@run line.substring(0, i)
            }
        }
        line
    }
    // Strip /* … */ block comments on a single line
    return noLineComment.replace(Regex("""/\*.*?\*/"""), "")
}

The test above is therefore a "doesn't exist yet" canary; the fix is the practice of always escaping, not a behavior assertion. Keep it as a documentation-style test.

}
```
